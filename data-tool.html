直接发送抓取到的数据文本即可，我会自动识别并更新报表。

格式示例：

企微数据： 周期: 04.24-04.30 会话量: 1525 派单工: 391 P4工: 216 电话数据： 周期: 04.24-04.30 呼入量: 327 接通率: 94%

📋 备用手动脚本

如果书签无法正常工作，可以在浏览器控制台（F12 → Console）中粘贴以下代码执行：

企微数据脚本：

javascript:(function(){const tables=document.querySelectorAll('table');let result='企微数据：\n';let weekData={sessions:0,orders:0,p4Orders:0};tables.forEach(table=>{const rows=table.querySelectorAll('tr');rows.forEach(row=>{const cells=row.querySelectorAll('td,th');if(cells.length>=2){const label=cells[0].innerText.trim();const value=cells[cells.length-1].innerText.trim();if(label.includes('会话量'))weekData.sessions=parseInt(value.replace(/[^0-9]/g,''))||0;if(label.includes('派单'))weekData.orders=parseInt(value.replace(/[^0-9]/g,''))||0;if(label.includes('P4'))weekData.p4Orders=parseInt(value.replace(/[^0-9]/g,''))||0;}});});result+=`会话量: ${weekData.sessions}\n派单工: ${weekData.orders}\nP4工: ${weekData.p4Orders}`;navigator.clipboard.writeText(result).then(()=>{alert('数据已复制！\n\n'+result);}).catch(()=>{prompt('请手动复制：',result);});})();

电话数据脚本：

javascript:(function(){const tables=document.querySelectorAll('table');let result='电话数据：\n';let phoneData={calls:0,connectRate:'0%'};tables.forEach(table=>{const rows=table.querySelectorAll('tr');rows.forEach(row=>{const cells=row.querySelectorAll('td,th');if(cells.length>=2){const label=cells[0].innerText.trim();const value=cells[cells.length-1].innerText.trim();if(label.includes('接入量')||label.includes('呼入量'))phoneData.calls=parseInt(value.replace(/[^0-9]/g,''))||0;if(label.includes('接通率')||label.includes('接通'))phoneData.connectRate=value;}});});result+=`呼入量: ${phoneData.calls}\n接通率: ${phoneData.connectRate}`;navigator.clipboard.writeText(result).then(()=>{alert('数据已复制！\n\n'+result);}).catch(()=>{prompt('请手动复制：',result);});})();
